cmake_minimum_required(VERSION 3.16)
project(rlinf_robotiq_observation LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(ROBOTIQ_WITH_ROS "Build the ROS 2 publisher and message tests" ON)
include(CTest)
find_package(Boost CONFIG REQUIRED COMPONENTS system)
find_package(Threads REQUIRED)
set(ROBOTIQ_SOURCE_DIR "/home/aist-franka/robotiq/robotiq-gripper-interface-main"
    CACHE PATH "Existing audited NUC library source (read only)")
include(cmake/vendor.cmake)
add_library(robotiq_vendor STATIC
  ${vendor_dir}/src/robotiq_gripper_interface.cc
  ${vendor_dir}/src/helpers.cc ${vendor_dir}/src/timeout_reader.cc)
target_include_directories(robotiq_vendor PUBLIC ${vendor_dir}/include PRIVATE ${vendor_dir})
target_link_libraries(robotiq_vendor PUBLIC Boost::system Threads::Threads)
add_library(observation_core STATIC src/observation.cpp src/command.cpp)
target_include_directories(observation_core PUBLIC include)
target_link_libraries(observation_core PUBLIC robotiq_vendor)
target_compile_options(observation_core PRIVATE -Wall -Wextra -Werror)
if(ROBOTIQ_WITH_ROS)
  find_package(ament_cmake REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(rcl_interfaces REQUIRED)
  find_package(diagnostic_msgs REQUIRED)
  find_package(builtin_interfaces REQUIRED)
  find_package(rosidl_default_generators REQUIRED)
  rosidl_generate_interfaces(${PROJECT_NAME}
    "srv/AuthorizeCommands.srv" "srv/SetNormalizedPosition.srv"
    DEPENDENCIES builtin_interfaces)
  rosidl_get_typesupport_target(command_typesupport ${PROJECT_NAME} rosidl_typesupport_cpp)
  add_library(observation_message STATIC src/message.cpp)
  target_link_libraries(observation_message observation_core)
  ament_target_dependencies(observation_message diagnostic_msgs)
  add_executable(robotiq_observation src/node.cpp)
  target_link_libraries(robotiq_observation observation_message)
  target_link_libraries(robotiq_observation "${command_typesupport}")
  ament_target_dependencies(robotiq_observation rclcpp rcl_interfaces)
  install(TARGETS robotiq_observation DESTINATION lib/${PROJECT_NAME})
  install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
  ament_export_dependencies(rosidl_default_runtime)
endif()
if(BUILD_TESTING)
  find_package(GTest REQUIRED)
  add_executable(test_observation test/test_observation.cpp)
  target_link_libraries(test_observation observation_core GTest::gtest_main util)
  target_include_directories(test_observation PRIVATE ${vendor_dir})
  if(ROBOTIQ_WITH_ROS)
    target_link_libraries(test_observation observation_message)
    target_compile_definitions(test_observation PRIVATE ROBOTIQ_TEST_ROS=1)
  endif()
  add_test(NAME robotiq_observation_cpu COMMAND test_observation)
  set_tests_properties(robotiq_observation_cpu PROPERTIES TIMEOUT 20)
  add_executable(test_command test/test_command.cpp)
  target_link_libraries(test_command observation_core GTest::gtest_main util)
  target_include_directories(test_command PRIVATE ${vendor_dir})
  if(ROBOTIQ_WITH_ROS)
    target_link_libraries(test_command observation_message "${command_typesupport}")
    ament_target_dependencies(test_command rclcpp)
    target_compile_definitions(test_command PRIVATE ROBOTIQ_TEST_ROS=1)
  endif()
  add_test(NAME robotiq_command_cpu COMMAND test_command)
  set_tests_properties(robotiq_command_cpu PROPERTIES TIMEOUT 20)
  if(ROBOTIQ_WITH_ROS)
    set_tests_properties(robotiq_command_cpu PROPERTIES ENVIRONMENT
      "LD_LIBRARY_PATH=$<TARGET_FILE_DIR:${command_typesupport}>:$ENV{LD_LIBRARY_PATH}")
  endif()
endif()
if(ROBOTIQ_WITH_ROS)
  ament_package()
endif()
